function renderDashboardKPIs() {
const totalMachineStock = products.reduce((acc, p) => acc + parseInt(p.stock), 0);
const totalSpareStock = spares.reduce((acc, s) => acc + parseInt(s.stock), 0);
// Calculate Financials (Estimated Sales from inventory)
const totalRev = products.reduce((acc, p) => acc + (p.price * 3), 145000); // Dummy accumulated revenue
const totalCost = products.reduce((acc, p) => acc + (p.costPrice * 3), 110000);
const netProfit = totalRev - totalCost;

const lowStockCount = products.filter(p => p.stock < 5).length + spares.filter(s => s.stock < 15).length;

document.getElementById('kpi-machine-stock').innerText = totalMachineStock;
document.getElementById('kpi-spare-stock').innerText = totalSpareStock;
document.getElementById('kpi-revenue').innerText = `₹${totalRev.toLocaleString()}`;
document.getElementById('kpi-profit').innerText = `₹${netProfit.toLocaleString()}`;
document.getElementById('kpi-enquiries').innerText = enquiries.filter(e => e.status === 'Pending').length;
document.getElementById('kpi-repairs').innerText = serviceBookings.filter(r => r.status === 'Pending').length;
document.getElementById('kpi-lowstock').innerText = lowStockCount;
}

const stock = parseInt(document.getElementById('prod-form-stock').value);
const img = document.getElementById('prod-form-img').value;
const desc = document.getElementById('prod-form-desc').value;

if (id) {
// Edit existing
const idx = products.findIndex(x => x.id === id);
if (idx !== -1) {
products[idx] = { ...products[idx], title, brand, model, category, costPrice, price, stock, img, desc };
}
} else {
// Add new
products.push({
id: 'p' + Date.now(),
title, brand, model, category, costPrice, price, stock, img, desc,
rating: 5.0, oldPrice: price + 2000
});
}

saveToStorage();
renderProducts();
renderOwnerProductsTable();
renderDashboardKPIs();
closeAddProductModal();
showToast('Inventory Updated', 'Product record saved successfully.');
}
